Query page layout

ListCard

UTable
Result list card with a title bar, a table and server-side pagination.

A list card for query results: a title bar, a ShuoTable data area and pagination, with pagination handled by the back end. The generic T is the row type. The skeleton, empty data, loading text and height transition are handled by ShuoTable; when you do not need the card and pagination, use ShuoTable directly.

Usage

Server-side pagination

Pagination is handled by the back end: the component emits page-change, and the page updates page and data after the query for that page succeeds. It is not emitted while loading, or when the page is the current one or out of range. interactionDisabled is true while loading and can be used to disable actions in the list.

<template>
  <ShuoListCard
    :data="rows"
    :columns="columns"
    :loading="loading"
    :total="total"
    :page="page"
    :page-size="5"
    @page-change="search"
  >
    <template #action-cell="{ row, interactionDisabled }">
      <UButton
        label="Details"
        size="xs"
        variant="soft"
        :disabled="interactionDisabled"
        @click="toast.add({ title: `Open ${row.original.name}` })"
      />
    </template>
  </ShuoListCard>
</template>

<script setup lang="ts">
import type { TableColumn } from '@nuxt/ui';
import type { PageRequest } from '#shuo-ui/types/pagination';

interface Row {
  id: number;
  name: string;
  amount: string;
}

const columns: TableColumn<Row>[] = [
  { accessorKey: 'id', header: 'No.' },
  { accessorKey: 'name', header: 'Name' },
  { accessorKey: 'amount', header: 'Amount' },
  { id: 'action', header: 'Actions' }
];
// Simulated back end: 23 rows, 5 per page; each query takes 0.6 seconds.
const all: Row[] = Array.from({ length: 23 }, (_, index) => ({
  id: index + 1,
  name: `Application ${index + 1}`,
  amount: ((index + 1) * 1250).toLocaleString()
}));
const rows = ref<Row[]>([]);
const total = ref(0);
const page = ref(1);
// Loading from creation, so the first query shows the table skeleton.
const loading = ref(true);
const toast = useToast();

// Update page and data only after the page loads; keep the existing data while loading.
const search = async ({ PageNumber, PageSize }: PageRequest) => {
  loading.value = true;
  await new Promise((resolve) => setTimeout(resolve, 600));
  rows.value = all.slice((PageNumber - 1) * PageSize, PageNumber * PageSize);
  total.value = all.length;
  page.value = PageNumber;
  loading.value = false;
};

onMounted(() => search({ PageNumber: 1, PageSize: 5 }));
</script>

The PageRequest type can be imported from #shuo-ui/types/pagination; see Auto-imports and types. For the full pattern with query fields, see With ListCard on ShuoQueryCard.

Pagination and counts

show-pagination controls the pagination at the bottom left, shown even with a single page; show-summary controls the count and range at the bottom right; page-size is the number of rows per page, 10 by default.

<template>
  <ExampleOptions>
    <ExampleOption
      v-model="showPagination"
      label="show-pagination"
      :items="[true, false]"
    />
    <ExampleOption
      v-model="showSummary"
      label="show-summary"
      :items="[true, false]"
    />
    <ExampleOption v-model="pageSize" label="page-size" :items="[3, 5, 10]" />
  </ExampleOptions>
  <ShuoListCard
    ref="card"
    :data="rows"
    :columns="columns"
    :loading="loading"
    :total="total"
    :page="page"
    :page-size="pageSize"
    :show-pagination="showPagination"
    :show-summary="showSummary"
    @page-change="search"
  />
</template>

<script setup lang="ts">
import type { TableColumn } from '@nuxt/ui';
import type { PageRequest } from '#shuo-ui/types/pagination';

interface Row {
  id: number;
  name: string;
}

const columns: TableColumn<Row>[] = [
  { accessorKey: 'id', header: 'No.' },
  { accessorKey: 'name', header: 'Name' }
];
// Simulated back end: 12 rows; each query takes 0.4 seconds.
const all: Row[] = Array.from({ length: 12 }, (_, index) => ({
  id: index + 1,
  name: `Application ${index + 1}`
}));
const showPagination = ref(true);
const showSummary = ref(true);
const pageSize = ref(5);
const card = useTemplateRef('card');
const rows = ref<Row[]>([]);
const total = ref(0);
const page = ref(1);
const loading = ref(false);

const search = async ({ PageNumber, PageSize }: PageRequest) => {
  loading.value = true;
  await new Promise((resolve) => setTimeout(resolve, 400));
  rows.value = all.slice((PageNumber - 1) * PageSize, PageNumber * PageSize);
  total.value = all.length;
  page.value = PageNumber;
  loading.value = false;
};

// After the page size changes, query again from page 1 with the new size.
watch(pageSize, async () => {
  await nextTick();
  const request = card.value?.getPagination(1);
  if (request) search(request);
});

onMounted(() => search({ PageNumber: 1, PageSize: pageSize.value }));
</script>
  • After changing page-size, query again from the first page with getPagination(1); see Re-querying the current page.
  • When both pagination and the summary are off, the bottom only shows #footer; without #footer, there is no bottom area.

Loading and empty data

When the component is created while loading with no data, skeleton rows are shown; when re-querying with existing data, the data fades and switches with a height transition. empty is the empty-data text, using the table default when not set; with transition set to false, no transition plays. The behavior is the same as ShuoTable.

<script setup lang="ts">
import type { TableColumn } from '@nuxt/ui';

interface Row {
  item: string;
  amount: string;
}

const columns: TableColumn<Row>[] = [
  { accessorKey: 'item', header: 'Item' },
  { accessorKey: 'amount', header: 'Amount' }
];
const source: Row[] = [
  { item: 'Mortgage', amount: '3,200,000' },
  { item: 'Personal loan', amount: '450,000' },
  { item: 'Car loan', amount: '680,000' }
];
const empty = ref('No data');
const transition = ref(true);
const rows = ref<Row[]>([]);
// Loading from creation, so the first load shows skeleton rows.
const loading = ref(true);

// Re-querying with data fades it and switches with a height transition; without data, loading text shows.
const reload = async () => {
  loading.value = true;
  await new Promise((resolve) => setTimeout(resolve, 800));
  rows.value = source.slice(0, Math.floor(Math.random() * 3) + 1);
  loading.value = false;
};

onMounted(async () => {
  await new Promise((resolve) => setTimeout(resolve, 1200));
  rows.value = [...source];
  loading.value = false;
});
</script>

The title "Results" and the bottom summary switch with the locale ("Results" and "Showing 1 to 10 of 25 results." in English, "查詢結果" and its summary in Traditional Chinese), and can be overridden globally with the listCard.title and listCard.summary texts; "Loading…" and "No data" belong to the table group. See Locales and messages.

import type { TableColumn } from '@nuxt/ui';

interface Row {
  id: string;
  applicant: string;
}

const columns: TableColumn<Row>[] = [
  { accessorKey: 'id', header: 'Case number' },
  { accessorKey: 'applicant', header: 'Applicant' }
];
const rows = ref<Row[]>([
  { id: 'A0001', applicant: 'Alex Wang' },
  { id: 'A0002', applicant: 'Mia Chen' }

const loading = ref(false);
// Get the time in the browser so the server and browser renders do not differ.
const updatedAt = ref('—');
const toast = useToast();

// While loading, interactionDisabled is true: the title bar buttons are disabled and the footer shows updating.
const reload = async () => {
  loading.value = true;
  await new Promise((resolve) => setTimeout(resolve, 800));
  updatedAt.value = new Date().toLocaleTimeString('en-US');
  loading.value = false;
};

onMounted(() => (updatedAt.value = new Date().toLocaleTimeString('en-US')));
</script>

Re-querying the current page

getPagination() on the template ref returns the current page number and the component's page size; passing a page number (such as getPagination(1)) returns that page instead, suited to the first query or going back to page one after the filters change.

<template>
  <ExampleOptions>
    <ExampleAction
      label="Re-query current page"
      icon="i-lucide-refresh-cw"
      :action="reload"
    />
  </ExampleOptions>
  <ShuoListCard
    ref="card"
    :data="rows"
    :columns="columns"
    :loading="loading"
    :total="total"
    :page="page"
    :page-size="4"
    @page-change="search"
  />
</template>

<script setup lang="ts">
import type { TableColumn } from '@nuxt/ui';
import type { PageRequest } from '#shuo-ui/types/pagination';

interface Row {
  id: number;
  status: string;
}

const columns: TableColumn<Row>[] = [
  { accessorKey: 'id', header: 'No.' },
  { accessorKey: 'status', header: 'Status' }
];
const statuses = ['In review', 'Approved', 'Returned'];
// Simulated back end: 10 rows with random statuses on each query; each query takes 0.6 seconds.
const fetchRows = () =>
  Array.from({ length: 10 }, (_, index) => ({
    id: index + 1,
    status: statuses[Math.floor(Math.random() * statuses.length)] ?? ''
  }));
const card = useTemplateRef('card');
const rows = ref<Row[]>([]);
const total = ref(0);
const page = ref(1);
const loading = ref(false);

const search = async ({ PageNumber, PageSize }: PageRequest) => {
  loading.value = true;
  await new Promise((resolve) => setTimeout(resolve, 600));
  const all = fetchRows();
  rows.value = all.slice((PageNumber - 1) * PageSize, PageNumber * PageSize);
  total.value = all.length;
  page.value = PageNumber;
  loading.value = false;
};

// Without a page number, the current page is used, with the component's page size.
const reload = async () => {
  const request = card.value?.getPagination();
  if (request) await search(request);
};

onMounted(() => {
  const request = card.value?.getPagination(1);
  if (request) search(request);
});
</script>
