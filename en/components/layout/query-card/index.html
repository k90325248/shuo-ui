Query page layout

QueryCard

UCardUForm
Collapsible query card containing a UForm, with fields in a responsive grid.

Fields are laid out in a responsive grid (one column, two from md, four from xl, adjustable with ui.grid). The generic S is the schema.

Usage

Title and icon

title is the title text and icon is the icon to the left of the title; pass an empty string to hide the icon.

Filters

<template>
  <ExampleOptions>
    <ExampleOption
      v-model="title"
      label="title"
      :items="['Filters', 'Case search', 'Advanced filters']"
    />
    <ExampleOption
      v-model="icon"
      label="icon"
      :items="['i-lucide-search', 'i-lucide-filter', 'i-lucide-file-search']"
    />
  </ExampleOptions>
  <ShuoQueryCard :state="query" :title="title" :icon="icon">
    <!-- Case number -->
    <UFormField label="Case number" name="caseNo">
      <ShuoInputText v-model="query.caseNo" />
    </UFormField>

    <!-- Applicant -->
    <UFormField label="Applicant" name="applicant">
      <ShuoInputText v-model="query.applicant" />
    </UFormField>
  </ShuoQueryCard>
</template>

<script setup lang="ts">
const title = ref('Filters');
const icon = ref('i-lucide-search');
const query = reactive({ caseNo: '', applicant: '' });
</script>

Collapsing

Control the expanded state with v-model:open, expanded by default. expand-icon is the button icon shown when collapsed, and collapse-icon is the button icon shown when expanded.

Filters

<template>
  <ExampleOptions>
    <ExampleOption v-model="open" label="open" :items="[true, false]" />
    <ExampleOption
      v-model="expandIcon"
      label="expand-icon"
      :items="['i-lucide-chevron-down', 'i-lucide-chevrons-down']"
    />
    <ExampleOption
      v-model="collapseIcon"
      label="collapse-icon"
      :items="['i-lucide-chevron-up', 'i-lucide-chevrons-up']"
    />
    <ExampleAction label="Submit" icon="i-lucide-send" :action="submit" />
  </ExampleOptions>
  <ShuoQueryCard
    ref="card"
    v-model:open="open"
    :state="query"
    :schema="schema"
    :expand-icon="expandIcon"
    :collapse-icon="collapseIcon"
    @submit="toast.add({ title: 'Search', description: query.name })"
  >
    <!-- Name -->
    <UFormField label="Name" name="name" required>
      <ShuoInputText v-model="query.name" />
    </UFormField>
  </ShuoQueryCard>
</template>

<script setup lang="ts">
import * as v from 'valibot';

const schema = v.object({ name: requiredText('Name') });
const query = reactive({ name: '' });
const open = ref(true);
const expandIcon = ref('i-lucide-chevron-down');
const collapseIcon = ref('i-lucide-chevron-up');
const card = useTemplateRef('card');
const toast = useToast();

// Submit while collapsed: with no name, validation fails and the card expands to show the error.
const submit = () => card.value?.submit();
</script>
  • When validation fails while collapsed, the card expands automatically to show the errors. In the example, collapse it first, then press "Submit" in the props bar to see it.
  • Collapsed fields cannot be focused or used.

Disabled

disabled disables the fields in the form (using the disabled state of UForm); submit() and clear() on the template ref and the submit and clear provided by #actions do nothing. Buttons in #actions are not disabled automatically, so pass disabled yourself.

Filters

<template>
  <ExampleOptions>
    <ExampleOption v-model="disabled" label="disabled" :items="[false, true]" />
  </ExampleOptions>
  <ShuoQueryCard
    :state="query"
    :disabled="disabled"
    @submit="toast.add({ title: 'Search', description: query.keyword })"
    @clear="query.keyword = ''"
  >
    <!-- Keyword -->
    <UFormField label="Keyword" name="keyword">
      <ShuoInputText v-model="query.keyword" />
    </UFormField>

    <template #actions="{ clear }">
      <UButton
        label="Clear"
        color="neutral"
        variant="soft"
        :disabled="disabled"
        @click="clear"
      />
      <UButton type="submit" label="Search" :disabled="disabled" />
    </template>
  </ShuoQueryCard>
</template>

<script setup lang="ts">
const disabled = ref(false);
const query = reactive({ keyword: 'Mortgage' });
const toast = useToast();
</script>

Columns

Adjust the number of columns with ui.grid. When changing columns, write out every breakpoint: tailwind-merge only replaces classes of the same breakpoint, so with only md:grid-cols-3, the built-in xl:grid-cols-4 still applies.

Filters

<template>
  <ExampleOptions>
    <ExampleOption v-model="layout" label="ui.grid" :items="layouts" />
  </ExampleOptions>
  <ShuoQueryCard :state="query" :ui="{ grid: GRIDS[layout] }">
    <!-- Query fields -->
    <UFormField v-for="(_, field) in query" :key="field" :label="field">
      <ShuoInputText v-model="query[field]" />
    </UFormField>
  </ShuoQueryCard>
</template>

<script setup lang="ts">
// Write out every breakpoint when changing columns; use complete static classes so Tailwind can scan them.
const GRIDS = {
  'built-in': undefined,
  '3 columns from md': 'md:grid-cols-3 xl:grid-cols-3',
  '6 columns from xl': 'md:grid-cols-3 xl:grid-cols-6'
};
const layouts = Object.keys(GRIDS) as (keyof typeof GRIDS)[];
const layout = ref<keyof typeof GRIDS>('built-in');

const query = reactive<Record<string, string>>({
  'Case number': '',
  'Applicant': '',
  'Handler': '',
  'Type': '',
  'Status': '',
  'Branch': ''
});
</script>

You can also remove the built-in breakpoints with the function form and set your own:

<ShuoQueryCard
  :state="query"
  :ui="{
    grid: (defaults) =>
      defaults.replace(/(md|xl):grid-cols-\d/g, '') + ' lg:grid-cols-3'
  }"
>
  ...
</ShuoQueryCard>

See Query field columns for site-wide columns.

Long titles

By default the title neither wraps nor truncates; when header-right does not fit, the whole block moves to the next line, aligned right, without squeezing the title or the collapse button. The title bar of ShuoListCard follows the same rules. To let a long title wrap next to the icon, override the title bar with ui so it does not wrap (bar: 'flex-nowrap') and allow the title to shrink (title: 'min-w-0 shrink whitespace-normal'):

Personal loan applications and disbursement progress filters

<template>
  <ExampleOptions>
    <ExampleOption v-model="length" label="title" :items="['short', 'long']" />
    <ExampleOption v-model="wrap" label="wrap title" :items="[false, true]" />
  </ExampleOptions>
  <!-- A narrow width shows the layout when header-right does not fit -->
  <div class="max-w-md">
    <ShuoQueryCard
      :state="query"
      :title="TITLES[length]"
      :ui="
        wrap
          ? { bar: 'flex-nowrap', title: 'min-w-0 shrink whitespace-normal' }
          : undefined
      "
    >
      <template #header-right>
        <UButton
          label="Export"
          icon="i-lucide-download"
          size="sm"
          variant="soft"
        />
      </template>

      <!-- Applicant -->
      <UFormField label="Applicant" name="applicant">
        <ShuoInputText v-model="query.applicant" />
      </UFormField>
    </ShuoQueryCard>
  </div>
</template>

<script setup lang="ts">
const TITLES = {
  short: 'Filters',
  long: 'Personal loan applications and disbursement progress filters'
};
const length = ref<keyof typeof TITLES>('long');
const wrap = ref(false);
const query = reactive({ applicant: '' });
</script>

Examples

Validation and action buttons

Press Enter in a field to submit; submit fires only after validation passes, with the data in event.data. Put the bottom buttons in #actions. clear fires the clear event; the page resets the data and the component then clears the validation errors.

Filters

<template>
  <ShuoQueryCard
    :state="query"
    :schema="schema"
    @submit="onSubmit"
    @clear="onClear"
  >
    <!-- Name -->
    <UFormField label="Name" name="name" required>
      <ShuoInputText v-model="query.name" />
    </UFormField>

    <!-- Type -->
    <UFormField label="Type" name="type">
      <ShuoSelectMenu
        v-model="query.type"
        :items="['Normal', 'Urgent']"
        placeholder="All"
      />
    </UFormField>

    <!-- Application date -->
    <UFormField label="Application date" name="date">
      <ShuoDatepicker v-model="query.date" clearable />
    </UFormField>

    <template #actions="{ clear }">
      <UButton label="Clear" color="neutral" variant="soft" @click="clear" />
      <UButton type="submit" label="Search" />
    </template>
  </ShuoQueryCard>
</template>

<script setup lang="ts">
import * as v from 'valibot';

const schema = v.object({
  name: requiredText('Name'),
  type: v.string(),
  date: v.string()
});
const query = reactive({ name: '', type: '', date: '' });
const toast = useToast();

// submit only fires after validation passes; on clear, the page resets the data and the component then clears the validation errors.
const onSubmit = (event: { data: v.InferOutput<typeof schema> }) => {
  toast.add({ title: 'Query', description: JSON.stringify(event.data) });
};
const onClear = () => {
  Object.assign(query, { name: '', type: '', date: '' });
};
</script>

Submitting from the title bar

Call submit() and clear() through the template ref, for example to put the search button in #header-right.

Filters

<template>
  <ShuoQueryCard
    ref="card"
    :state="query"
    :schema="schema"
    @submit="onSubmit"
    @clear="query.caseNo = ''"
  >
    <template #header-right>
      <UButton
        label="Clear"
        icon="i-lucide-rotate-ccw"
        size="sm"
        color="neutral"
        variant="soft"
        @click="card?.clear()"
      />
      <UButton
        label="Search"
        icon="i-lucide-search"
        size="sm"
        @click="card?.submit()"
      />
    </template>

    <!-- Case number -->
    <UFormField label="Case number" name="caseNo" required>
      <ShuoInputText v-model="query.caseNo" placeholder="For example A0001" />
    </UFormField>
  </ShuoQueryCard>
</template>

<script setup lang="ts">
import * as v from 'valibot';

const schema = v.object({ caseNo: requiredText('Case number') });
const query = reactive({ caseNo: '' });
const card = useTemplateRef('card');
const toast = useToast();

const onSubmit = (event: { data: v.InferOutput<typeof schema> }) => {
  toast.add({ title: 'Search case', description: event.data.caseNo });
};
</script>

With ListCard

After submitting the query, call getPagination(1) on ShuoListCard to query from the first page; changing pages keeps the submitted query.

Filters

Results

No.Name

Showing 0 to 0 of 0 results.

<template>
  <div class="flex flex-col gap-4">
    <ShuoQueryCard
      :state="query"
      @submit="onSubmit"
      @clear="query.keyword = ''"
    >
      <!-- Name keyword -->
      <UFormField label="Name keyword" name="keyword">
        <ShuoInputText v-model="query.keyword" placeholder="For example 3" />
      </UFormField>

      <template #actions="{ clear }">
        <UButton label="Clear" color="neutral" variant="soft" @click="clear" />
        <UButton type="submit" label="Search" :loading="loading" />
      </template>
    </ShuoQueryCard>
    <ShuoListCard
      ref="list"
      :data="rows"
      :columns="columns"
      :loading="loading"
      :total="total"
      :page="page"
      :page-size="5"
      @page-change="search"
    />
  </div>
</template>

<script setup lang="ts">
import type { TableColumn } from '@nuxt/ui';
import type { PageRequest } from '#shuo-ui/types/pagination';

interface Row {
  id: number;
  name: string;
}

const columns: TableColumn<Row>[] = [
  { accessorKey: 'id', header: 'No.' },
  { accessorKey: 'name', header: 'Name' }
];
// Simulated back end: 23 rows, filtered by keyword then paginated; each query takes 0.6 seconds.
const all: Row[] = Array.from({ length: 23 }, (_, index) => ({
  id: index + 1,
  name: `Application ${index + 1}`
}));
const query = reactive({ keyword: '' });
// The submitted keyword; changing pages keeps it, unaffected by input not yet submitted.
const applied = ref('');
const list = useTemplateRef('list');
const rows = ref<Row[]>([]);
const total = ref(0);
const page = ref(1);
// Loading from creation, so the first query shows the table skeleton.
const loading = ref(true);

const search = async ({ PageNumber, PageSize }: PageRequest) => {
  loading.value = true;
  await new Promise((resolve) => setTimeout(resolve, 600));
  const matched = all.filter((row) => row.name.includes(applied.value));
  rows.value = matched.slice(
    (PageNumber - 1) * PageSize,
    PageNumber * PageSize
  );
  total.value = matched.length;
  page.value = PageNumber;
  loading.value = false;
};

// Submitting the query goes back to page 1, with the page size from the list card.
const onSubmit = () => {
  applied.value = query.keyword;
  const request = list.value?.getPagination(1);
  if (request) search(request);
};

onMounted(() => search({ PageNumber: 1, PageSize: 5 }));
</script>
